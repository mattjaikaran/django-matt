   ```python
   class assert_query_count:
       """Context manager and decorator for asserting database query counts.

       Usage as context manager:
           with assert_query_count(3):
               list(MyModel.objects.all())

       Usage as decorator:
           @assert_query_count(3)
           def test_something():
               list(MyModel.objects.all())
       """

       def __init__(self, expected: int, using: str = "default"):
           self.expected = expected
           self.using = using

       def __enter__(self):
           from django.db import connections
           from django.test.utils import CaptureQueriesContext
           conn = connections[self.using]
           self._ctx = CaptureQueriesContext(conn)
           self._ctx.__enter__()
           return self

       def __exit__(self, exc_type, exc_val, exc_tb):
           self._ctx.__exit__(exc_type, exc_val, exc_tb)
           if exc_type is None:
               actual = len(self._ctx.captured_queries)
               if actual != self.expected:
                   queries = "\n".join(
                       f"  {i+1}. {q['sql']}"
                       for i, q in enumerate(self._ctx.captured_queries)
                   )
                   raise AssertionError(
                       f"Expected {self.expected} queries, got {actual}.\n"
                       f"Queries:\n{queries}"
                   )

       @property
       def captured_queries(self):
           """Access captured queries (available after context exits)."""
           return self._ctx.captured_queries

       def __call__(self, func):
           """Use as decorator."""
           import functools
           @functools.wraps(func)
           def wrapper(*args, **kwargs):
               with assert_query_count(self.expected, self.using):
                   return func(*args, **kwargs)
           return wrapper
   ```
   NOTE: Use `connections[self.using]` (not bare `connection`) to support multi-db setups.

3. Add `cache_response` decorator to `django_matt/utils/performance.py`:
   ```python
   def cache_response(timeout: int = 300, key_prefix: str = "matt_cache"):
       """Cache view method response for `timeout` seconds.

       Usage:
           class MyController(APIController):
               @api.get("/items")
               @cache_response(timeout=300)
               async def list_items(self, request):
                   return await self.get_queryset()
       """
       def decorator(func):
           @functools.wraps(func)
           async def async_wrapper(self_or_request, *args, **kwargs):
               # Determine request object
               request = args[0] if args else kwargs.get("request")
               if request is None and hasattr(self_or_request, "method"):
                   request = self_or_request
               elif request is None:
                   # self_or_request is `self`, first arg is request
                   request = args[0] if args else None

               if request is not None:
                   cache_key = f"{key_prefix}:{request.path}:{request.GET.urlencode()}"
                   cache_key = hashlib.md5(cache_key.encode()).hexdigest()
                   cached = django_cache.get(cache_key)
                   if cached is not None:
                       return cached

               result = await func(self_or_request, *args, **kwargs)

               if request is not None:
                   django_cache.set(cache_key, result, timeout)
               return result

           @functools.wraps(func)
           def sync_wrapper(self_or_request, *args, **kwargs):
               request = args[0] if args else kwargs.get("request")
               if request is None and hasattr(self_or_request, "method"):
                   request = self_or_request

               if request is not None:
                   cache_key = f"{key_prefix}:{request.path}:{request.GET.urlencode()}"
                   cache_key = hashlib.md5(cache_key.encode()).hexdigest()
                   cached = django_cache.get(cache_key)
                   if cached is not None:
                       return cached

               result = func(self_or_request, *args, **kwargs)

               if request is not None:
                   django_cache.set(cache_key, result, timeout)
               return result

           import asyncio
           if asyncio.iscoroutinefunction(func):
               return async_wrapper
           return sync_wrapper
       return decorator
   ```
